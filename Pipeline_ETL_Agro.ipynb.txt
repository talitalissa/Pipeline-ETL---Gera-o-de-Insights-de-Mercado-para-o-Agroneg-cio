        "## 2. Transformação = Aqui reside a lógica de negócios e inteligência analítica do pipeline. Mapeamos cada perfil produtivo e volume de giro (em arrobas) para aplicar regras condicionais que simulam um motor de decisão inteligente, gerando estratégias personalizadas de proteção de mercado (hedge e custos).\n",
        "\n",
        "def gerar_insight_mercado(linha):\n",
        "    nome = linha['Nome']\n",
        "    tipo = linha['Tipo_Producao']\n",
        "    giro = linha['Giro_Anual_Arrobas']\n",
        "\n",
        "    # Regras de negócio associadas ao mercado pecuário\n",
        "    if tipo == \"Confinamento Tecnificado\":\n",
        "        return f\"Olá, {nome}! Com base no seu giro de {giro} @, o mercado futuro indica excelente oportunidade para travar preços de segundo giro via contratos de opções na B3. Proteja sua margem contra a volatilidade do boi gordo.\"\n",
        "    elif tipo == \"Recria e Engorda\":\n",
        "        return f\"Olá, {nome}! Momento estratégico para monitorar a relação de troca entre o boi gordo e o bezerro. Recomendamos avaliar o custo de reposição neste trimestre para otimizar suas margens.\"\n",
        "    else: # Cria\n",
        "        return f\"Olá, {nome}! A retenção de matrizes mostra-se favorável neste ciclo pecuário. Avalie ferramentas de crédito rural para sustentar o manejo até o momento ideal de comercialização dos bezerros.\"\n",
        "\n",
        "def transformar_dados(df):\n",
        "    print(\"=== Iniciando Etapa de Transformação ===\")\n",
        "    # Criando a nova coluna de insights personalizados baseados nas regras\n",
        "    df['Mensagem_Marketing'] = df.apply(gerar_insight_mercado, axis=1)\n",
